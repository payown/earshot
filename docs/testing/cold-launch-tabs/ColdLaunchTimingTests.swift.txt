import XCTest

@MainActor
final class ColdLaunchTimingTests: XCTestCase {
    func testLaunchAndFirstTabCycle() {
        let app = XCUIApplication()
        app.launchArguments = ["-uiTestScreenshotSeed", "-screenshotScreen", "library"]
        for sample in 0..<5 {
            app.terminate()
            let start = ProcessInfo.processInfo.systemUptime
            app.launch()
            XCTAssertTrue(app.buttons["Library"].firstMatch.waitForExistence(timeout: 15))
            let launch = ProcessInfo.processInfo.systemUptime - start
            let tabStart = ProcessInfo.processInfo.systemUptime
            for name in ["Inbox", "Queue", "Downloads", "Settings", "Library"] {
                let tab = app.buttons[name].firstMatch
                XCTAssertTrue(tab.waitForExistence(timeout: 10))
                tab.tap()
                XCTAssertTrue(tab.isSelected)
            }
            print("COLDTAB_UI sample=\(sample) launch_ms=\(launch * 1000) first_tab_cycle_ms=\((ProcessInfo.processInfo.systemUptime - tabStart) * 1000)")
        }
        app.terminate()
    }
}
